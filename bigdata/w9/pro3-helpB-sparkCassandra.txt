SPARK - CASSANDRA:

1) Start Cassandra using systemctl start cassandra

2) Start Spark with script provided under hadoop account

3) Start spark-shell with cassandra driver

> spark-shell --packages com.datastax.spark:spark-cassandra-connector_2.12:3.3.0 

4) Run following statements:

sc.stop()

import com.datastax.spark.connector._
import org.apache.spark.SparkContext
import org.apache.spark.SparkContext._
import org.apache.spark.SparkConf

val conf = new SparkConf(true).set("spark.cassandra.connection.host", "127.0.0.1")

val sc = new SparkContext("local[2]", "Connection test", conf)

val empRDD = sc.cassandraTable("test", "emp")

empRDD.count()

empRDD.collect().foreach(println)

empRDD.take(5).foreach(println)

empRDD.select("empno").collect().foreach(println)

case class Emp(empno: Int, ename: String, sal: Double)
 
val rec = sc.parallelize(Seq(Emp(10,"Ahmet",34.12)))

rec.saveToCassandra("test","emp", SomeColumns("empno","ename","sal"))

val empRDD1 = sc.cassandraTable("test", "emp")

empRDD1.collect().foreach(println)
